if event.type == "program" then
    mem.stars = {}
    for i = 1, 12 do
        table.insert(mem.stars, {x = math.random(1, 64), y = math.random(1, 64), s = math.random(1, 2)})
    end
    mem.y = 28
    mem.dir = 1
    mem.laser_x = 0
    mem.enemy_x = 64
    mem.enemy_y = 28
    mem.boom = 0
    interrupt(0.04)
end

if event.type == "interrupt" then
    local cmds = {{command = "createbuffer", buffer = 0, xsize = 64, ysize = 64, fill = "020208"}}

    for i = 1, #mem.stars do
        local s = mem.stars[i]
        s.x = s.x - s.s
        if s.x < 1 then s.x = 64 s.y = math.random(1, 64) end
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = s.x, y1 = s.y, x2 = s.x, y2 = s.y, fill = s.s == 2 and "aaaaaa" or "444455"})
    end

    mem.y = mem.y + mem.dir
    if mem.y > 45 or mem.y < 5 then mem.dir = -mem.dir end

    if mem.laser_x > 0 then
        mem.laser_x = mem.laser_x + 3
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = mem.laser_x, y1 = mem.laser_y, x2 = mem.laser_x + 3, y2 = mem.laser_y, fill = "00ffff"})
        if mem.laser_x > 64 then mem.laser_x = 0 end
    else
        if math.random(1, 8) == 1 then
            mem.laser_x = 14
            mem.laser_y = mem.y + 3
        end
    end

    if mem.boom > 0 then
        mem.boom = mem.boom - 1
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = math.floor(mem.enemy_x) - 4, y1 = mem.enemy_y - 4, x2 = math.floor(mem.enemy_x) + 8, y2 = mem.enemy_y + 8, fill = "ff5500"})
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = math.floor(mem.enemy_x) - 1, y1 = mem.enemy_y - 1, x2 = math.floor(mem.enemy_x) + 5, y2 = mem.enemy_y + 5, fill = "ffff00"})
        if mem.boom == 0 then
            mem.enemy_x = 64
            mem.enemy_y = math.random(10, 44)
        end
    else
        mem.enemy_x = mem.enemy_x - 1
        if mem.enemy_x < -8 then
            mem.enemy_x = 64
            mem.enemy_y = math.random(10, 44)
        end

        if mem.laser_x > 0 and mem.laser_x >= mem.enemy_x and mem.laser_x <= mem.enemy_x + 6 and math.abs(mem.laser_y - (mem.enemy_y + 2)) <= 4 then
            mem.boom = 5
            mem.laser_x = 0
        end

        local ex = math.floor(mem.enemy_x)
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = ex, y1 = mem.enemy_y + 1, x2 = ex + 5, y2 = mem.enemy_y + 4, fill = "ff3333"})
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = ex + 2, y1 = mem.enemy_y, x2 = ex + 4, y2 = mem.enemy_y + 5, fill = "aa0000"})
    end

    local sy = mem.y
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = 4, y1 = sy + 2, x2 = 12, y2 = sy + 4, fill = "aaaaaa"})
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = 1, y1 = sy, x2 = 3, y2 = sy + 6, fill = "555555"})
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = 6, y1 = sy + 1, x2 = 9, y2 = sy + 5, fill = "00aaff"})

    if math.random(1, 2) == 1 then
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = 1, y1 = sy + 1, x2 = 1, y2 = sy + 1, fill = "ffaa00"})
        table.insert(cmds, {command = "drawrect", buffer = 0, x1 = 1, y1 = sy + 5, x2 = 1, y2 = sy + 5, fill = "ffaa00"})
    end

    table.insert(cmds, {command = "send", buffer = 0, channel = "screen", screen_width = 3, screen_height = 2})
    digiline_send("gpu", cmds)
    interrupt(0.04)
end
