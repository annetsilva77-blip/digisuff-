if event.type == "program" then
    mem.stars = {}
    for i = 1, 20 do
        table.insert(mem.stars, {
            x = math.random(1, 64),
            y = math.random(1, 48),
            speed = math.random(1, 3)
        })
    end
    mem.ship_x = 28
    mem.ship_dir = 1
    mem.laser_y = 0
    mem.laser_x = 0
    interrupt(0.1)
end

if event.type == "interrupt" then
    mem.ship_x = mem.ship_x + mem.ship_dir
    if mem.ship_x >= 48 or mem.ship_x <= 2 then
        mem.ship_dir = -mem.ship_dir
    end

    if mem.laser_y > 0 then
        mem.laser_y = mem.laser_y - 4
    else
        if math.random(1, 5) == 1 then
            mem.laser_x = mem.ship_x + 7
            mem.laser_y = 48
        end
    end

    local cmds = {
        {command = "createbuffer", buffer = 0, xsize = 64, ysize = 64, fill = "020208"}
    }

    for i = 1, #mem.stars do
        local s = mem.stars[i]
        s.y = s.y + s.speed
        if s.y > 48 then
            s.y = 1
            s.x = math.random(1, 64)
            s.speed = math.random(1, 3)
        end
        
        local color = "555555"
        if s.speed == 2 then color = "aaaaaa" end
        if s.speed == 3 then color = "ffffff" end
        
        table.insert(cmds, {
            command = "drawrect",
            buffer = 0,
            x1 = s.x,
            y1 = s.y,
            x2 = s.x,
            y2 = s.y,
            fill = color
        })
    end

    if mem.laser_y > 0 then
        table.insert(cmds, {
            command = "drawline",
            buffer = 0,
            x1 = mem.laser_x,
            y1 = mem.laser_y,
            x2 = mem.laser_x,
            y2 = mem.laser_y + 4,
            color = "00ffff"
        })
    end

    local sx = mem.ship_x
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = sx + 2, y1 = 54, x2 = sx + 12, y2 = 58, fill = "888888"})
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = sx + 5, y1 = 51, x2 = sx + 9, y2 = 53, fill = "aaaaaa"})
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = sx, y1 = 56, x2 = sx + 1, y2 = 59, fill = "ff3333"})
    table.insert(cmds, {command = "drawrect", buffer = 0, x1 = sx + 13, y1 = 56, x2 = sx + 14, y2 = 59, fill = "ff3333"})

    if math.random(1, 2) == 1 then
        table.insert(cmds, {command = "drawline", buffer = 0, x1 = sx + 3, y1 = 59, x2 = sx + 3, y2 = 62, color = "ffaa00"})
        table.insert(cmds, {command = "drawline", buffer = 0, x1 = sx + 11, y1 = 59, x2 = sx + 11, y2 = 62, color = "ffaa00"})
    end

    table.insert(cmds, {command = "send", buffer = 0, channel = "screen"})
    digiline_send("gpu", cmds)
    interrupt(0.1)
end
